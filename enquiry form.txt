<script>
(function () {
  // ⚠️ Yaha apna Google Apps Script "Web app URL" paste karo (/exec par khatam hone wala)
  var SCRIPT_URL = "https://script.google.com/macros/s/AKfycbwaN5nKPomrqZJIy38qEd-ToPLzneDhfM-MEmzIVHCn0ul78BTkRFhk_DzbayKFx9GY/exec";

  var form = document.getElementById("smeForm");
  var btn = document.getElementById("smeBtn");
  var msg = document.getElementById("smeMsg");

  var stars = document.querySelectorAll("#smeStars .sme-star");
  var rating = 0;
  function paint(n) {
    stars.forEach(function (s) { s.classList.toggle("on", Number(s.dataset.v) <= n); });
  }
  stars.forEach(function (s) {
    s.addEventListener("click", function () { rating = Number(s.dataset.v); paint(rating); });
    s.addEventListener("mouseenter", function () { paint(Number(s.dataset.v)); });
  });
  document.getElementById("smeStars").addEventListener("mouseleave", function () { paint(rating); });

  function show(text, cls) { msg.className = "sme-msg " + (cls || ""); msg.textContent = text; }

  form.addEventListener("submit", function (e) {
    e.preventDefault();

    var name = form.name.value.trim();
    var company = form.company.value.trim();
    var email = form.email.value.trim();
    var requirement = form.requirement.value.trim();
    var digits = form.mobile.value.replace(/\D/g, "");
    if (digits.length === 12 && digits.indexOf("91") === 0) digits = digits.slice(2);

    if (!name) return show("Apna naam likhein.", "err");
    if (!company) return show("Company ka naam likhein.", "err");
    if (!/^[6-9]\d{9}$/.test(digits)) return show("Sahi 10 digit ka mobile number daalein.", "err");
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) return show("Sahi e-mail daalein.", "err");
    if (!requirement) return show("Apni requirement likhein.", "err");
    if (form.website.value) return; // bot
    if (SCRIPT_URL.indexOf("PASTE_YOUR") !== -1) return show("SCRIPT_URL abhi set nahi hai.", "err");

    btn.disabled = true;
    btn.textContent = "Bhej rahe hain...";
    show("");

    fetch(SCRIPT_URL, {
      method: "POST",
      mode: "no-cors",
      headers: { "Content-Type": "text/plain;charset=utf-8" },
      body: JSON.stringify({ name: name, company: company, mobile: digits, email: email, requirement: requirement, rating: rating || "" })
    }).then(function () {
      show("Dhanyavaad! Aapki enquiry mil gayi hai. Hum jaldi sampark karenge.", "ok");
      form.reset();
      rating = 0;
      paint(0);
    }).catch(function () {
      show("Submit nahi hua. Internet check karke dobara koshish karein.", "err");
    }).finally(function () {
      btn.disabled = false;
      btn.textContent = "Submit Enquiry →";
    });
  });
})();
</script>
